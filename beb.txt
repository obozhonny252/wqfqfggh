local posCache = setmetatable({}, {__mode = "k"})

local function getPosSizeRaw(obj)
    if not obj or not obj.Parent then return nil end
    if obj:IsA("BasePart") then
        return obj.Position, obj.Size.Magnitude
    elseif obj:IsA("Model") then
        local prim = obj.PrimaryPart
        if prim then return prim.Position, prim.Size.Magnitude end
        local ok, pivot = pcall(function() return obj:GetPivot() end)
        if ok and pivot then return pivot.Position, 10 end
        return nil
    end
    return nil
end

local function getPosSize(obj)
    local c = posCache[obj]
    local now = os.clock()
    if c and now - c.t < CONFIG.posCacheTTL then
        return c.pos, c.size
    end
    local pos, size = getPosSizeRaw(obj)
    if pos then
        posCache[obj] = {pos = pos, size = size, t = now}
    end
    return pos, size
end

-- ============================================================
-- BLACKLIST + COLLECT
-- ============================================================
local function hasBlacklistedName(name)
    if type(name) ~= "string" then return true end
    local l = string.lower(name)
    for _, w in ipairs(CONFIG.droneKeywords) do
        if string.find(l, w, 1, true) then return false end
    end
    for _, w in ipairs(CONFIG.blacklistKeywords) do
        if string.find(l, w, 1, true) then return true end
    end
    return false
end

local function collectDescendants(root, maxDepth, maxCount, applyBlacklist)
    maxDepth = maxDepth or CONFIG.nearbyDepth
    maxCount = maxCount or CONFIG.scanMaxCount
    local result = {}
    local count = 0
    local function walk(node, depth)
        if count >= maxCount then return end
        local children = node:GetChildren()
        for i = 1, #children do
            if count >= maxCount then return end
            local child = children[i]
            if applyBlacklist and hasBlacklistedName(child.Name) then
                -- skip whole subtree
            else
                count = count + 1
                result[count] = child
                if depth < maxDepth then
                    walk(child, depth + 1)
                end
            end
        end
    end
    walk(root, 1)
    return result
end

-- ============================================================
-- GROUND / AUTO HEIGHT
-- ============================================================
local function getGroundY()
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then return nil end

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {char}

    local result = workspace:Raycast(root.Position, Vector3.new(0, -500, 0), params)
    if result then return result.Position.Y end
    return nil
end

local function updateAutoHeight()
    if not CONFIG.autoHeight then return end
    local gY = getGroundY()
    if not gY then return end
    CONFIG.groundY = gY - 5
    CONFIG.flyMinY = gY + 8
end

-- ============================================================
-- HELPERS
-- ============================================================
local function classAllowed(obj)
    return CONFIG.allowedClasses[obj.ClassName] == true
end

local function isLikelyDrone(obj)
    if not classAllowed(obj) then return false end
    if hasBlacklistedName(obj.Name) then return false end
    local pos, size = getPosSize(obj)
    if not pos or not size then return false end
    if size < CONFIG.minSizeMagnitude then return false end
    if size > CONFIG.maxSizeMagnitude then return false end
    if pos.Y < CONFIG.flyMinY then return false end
    return true
end

local function nameMatchesPattern(name, p)
    if p.match == "exact" then
        return name == p.id
    elseif p.match == "contains" then
        return string.find(string.lower(name), string.lower(p.id), 1, true) ~= nil
    else
        return string.lower(string.sub(name, 1, #p.id)) == string.lower(p.id)
    end
end

local ownVehicleCache = {t = 0, val = nil}

local function getOwnVehicle()
    local now = os.clock()
    if now - ownVehicleCache.t < 0.4 then return ownVehicleCache.val end
    ownVehicleCache.t = now

    local char = LocalPlayer.Character

    if char then
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum and hum.SeatPart then
            local parent = hum.SeatPart.Parent
            local v = (parent and parent ~= char and parent ~= workspace) and parent or hum.SeatPart
            ownVehicleCache.val = v
            return v
        end

        for _, d in ipairs(char:GetDescendants()) do
            if d:IsA("Weld") or d:IsA("WeldConstraint") or d:IsA("Motor6D") then
                local p0, p1 = d.Part0, d.Part1
                local other = nil
                if p1 and not p1:IsDescendantOf(char) then other = p1 end
                if not other and p0 and not p0:IsDescendantOf(char) then other = p0 end
                if other then
                    local par = other.Parent
                    local v = (par and par:IsA("Model")) and par or other
                    ownVehicleCache.val = v
                    return v
                end
            end
        end
    end

    local cam = workspace.CurrentCamera
    if cam then
        local subj = cam.CameraSubject

        if subj and subj:IsA("BasePart") then
            local insideChar = char and subj:IsDescendantOf(char)
            if not insideChar then
                local par = subj.Parent
                local v = (par and par:IsA("Model")) and par or subj
                ownVehicleCache.val = v
                return v
            end
        end

        if subj and subj:IsA("Attachment") then
            local p = subj.Parent
            while p and p ~= workspace do
                if p:IsA("Model") then
                    ownVehicleCache.val = p
                    return p
                end
                p = p.Parent
            end
            ownVehicleCache.val = subj.Parent
            return subj.Parent
        end
    end

    ownVehicleCache.val = nil
    return nil
end

local playerPosCache = {t = 0, val = nil}
local function getPlayerPos()
    local now = os.clock()
    if now - playerPosCache.t < 0.15 then return playerPosCache.val end
    playerPosCache.t = now

    local pos
    local veh = getOwnVehicle()
    if veh then pos = getPosSize(veh) end
    if not pos then
        local char = LocalPlayer.Character
        if char then
            local root = char:FindFirstChild("HumanoidRootPart")
            if root then pos = root.Position end
        end
    end
    if not pos then
        local cam = workspace.CurrentCamera
        if cam then pos = cam.CFrame.Position end
    end

    playerPosCache.val = pos
    return pos
end

-- ============================================================
-- IS DESTROYED
-- ============================================================
local function isDestroyed(obj)
    if not obj.Parent or not obj:IsDescendantOf(workspace) then return true end

    local pos = getPosSize(obj)
    if pos and pos.Y < CONFIG.groundY then return true end

    if obj:IsA("Model") then
        local hum = obj:FindFirstChildOfClass("Humanoid")
        if hum and hum.Health <= 0 then return true end

        local ok, v = pcall(function() return obj:GetAttribute("Destroyed") end)
        if ok and v == true then return true end
        local ok2, v2 = pcall(function() return obj:GetAttribute("dead") end)
        if ok2 and v2 == true then return true end

        if obj:HasTag("Destroyed") or obj:HasTag("Dead") then return true end
    end
    return false
end

-- ============================================================
-- HIGHLIGHT
-- ============================================================
local function removeHighlightFrom(inst)
    if not inst then return end
    local hl = inst:FindFirstChild("OT_HL")
    if hl then hl:Destroy() end
    local own = inst:FindFirstChild("OT_OwnHL")
    if own then own:Destroy() end
end

local function createHighlightFor(inst, isOwn)
    local name = isOwn and "OT_OwnHL" or "OT_HL"
    local hl = inst:FindFirstChild(name)
    if not hl then
        hl = Instance.new("Highlight")
        hl.Adornee = inst
        hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        hl.Name = name
        hl.Parent = inst
    end
    if isOwn then
        hl.FillTransparency = CONFIG.ownFillTransparency
        hl.OutlineColor = CONFIG.ownOutlineColor
        hl.OutlineTransparency = 0
    else
        hl.FillTransparency = CONFIG.fillTransparency
        hl.OutlineColor = CONFIG.outlineColor
        hl.OutlineTransparency = CONFIG.outlineTransparency
    end
    hl.Enabled = true
    return hl
end

local function addTracked(inst, name)
    if tracked[inst] then return false end
    local count = 0
    for _ in pairs(tracked) do count = count + 1 end
    if count >= CONFIG.maxTracked then return false end
    tracked[inst] = {
        displayName = name or inst.Name,
        isActive    = false,
        highlight   = nil,
    }
    return true
end

local function removeTracked(inst)
    if tracked[inst] then
        removeHighlightFrom(inst)
        tracked[inst] = nil
    end
end

    if not distGui then return end
    if not CONFIG.showDistanceLabels then
        for i = 1, #labelPool do
            local l = labelPool[i]
            if l.Visible then l.Visible = false end
        end
        return
    end
    local cam = workspace.CurrentCamera
    if not cam then return end
    local camPos = cam.CFrame.Position

    local idx = 1
    for inst, data in pairs(tracked) do
        if data.isActive and idx <= #labelPool then
            local pos = getPosSizeRaw(inst)
            if pos then
                local screen = cam:WorldToViewportPoint(pos)
                if screen.Z > 0 then
                    local dist = (pos - camPos).Magnitude
                    local nm = tostring(data.displayName or inst.Name or "?")
                    local lbl = labelPool[idx]
                    lbl.Position = UDim2.fromOffset(screen.X, screen.Y - CONFIG.labelOffsetY)
                    lbl.Text = string.format("%s  %.0fm", nm, dist)
                    if not lbl.Visible then lbl.Visible = true end
                    idx = idx + 1
                end
            end
        end
    end
    for i = idx, #labelPool do
        local l = labelPool[i]
        if l.Visible then l.Visible = false end
    end
end)

local function doNearbyScan()
    task.spawn(function()
        local pp = getPlayerPos()
        if not pp then return end
        local radius = CONFIG.nearbyRadius
        local all = collectDescendants(workspace, CONFIG.nearbyDepth,
                                       CONFIG.scanMaxCount, true)
        local rows = {}
        for _, obj in ipairs(all) do
            if obj:IsA("BasePart") or obj:IsA("Model") then
                local pos, size = getPosSize(obj)
                if pos and size then
                    local d = (pos - pp).Magnitude
                    if d <= radius then
                        table.insert(rows, {
                            name=obj.Name, class=obj.ClassName,
                            dist=d, y=pos.Y, size=size,
                        })
                    end
                end
            end
        end
        table.sort(rows, function(a,b) return a.dist < b.dist end)
        local lines = {}
        for i = 1, math.min(#rows, 40) do
            local r = rows[i]
            table.insert(lines, string.format("%s|%s|%.0fm|Y=%.0f|S=%.1f",
                r.class, r.name, r.dist, r.y, r.size))
        end
        if #rows == 0 then lines = {"Nothing."} end
        pcall(function()
            NearbyParagraph:Set({
                Title = "Nearby ("..#rows..") r="..radius.." scanned="..#all,
                Content = table.concat(lines,"\n"),
            })
        end)
    end)
end

-- ============================================================
-- OWN HIGHLIGHT
-- ============================================================
local ownHighlighted = nil
local function updateOwnHighlight()
    local target = getOwnVehicle()
    if not target then
        if ownHighlighted then
            removeHighlightFrom(ownHighlighted)
            ownHighlighted = nil
        end
        return
    end
    if ownHighlighted ~= target or not target:FindFirstChild("OT_OwnHL") then
        if ownHighlighted and ownHighlighted ~= target then
            removeHighlightFrom(ownHighlighted)
        end
        ownHighlighted = target
        createHighlightFor(target, true)
    end
end

-- ============================================================
-- REFRESH HIGHLIGHTS
-- ============================================================
local function refreshHighlights()
    local pp = getPlayerPos()
    if not pp then return end

    local list = {}
    for inst, data in pairs(tracked) do
        if not inst.Parent or not inst:IsDescendantOf(workspace) then
            removeTracked(inst)
        else
            local pos = getPosSize(inst)
            if pos then
                local dist = (pos - pp).Magnitude
                if dist > CONFIG.maxTrackedDistance then
                    removeTracked(inst)
                elseif not isDestroyed(inst) and dist <= currentDistance then
                    table.insert(list, {inst = inst, dist = dist})
                end
            end
        end
    end
    table.sort(list, function(a,b) return a.dist < b.dist end)

    local activeSet = {}
    local limit = math.min(#list, CONFIG.maxActiveHighlights)
    for i = 1, limit do activeSet[list[i].inst] = true end

    for inst, data in pairs(tracked) do
        if not inst.Parent or not inst:IsDescendantOf(workspace) then
            removeTracked(inst)
        else
            local shouldBe = activeSet[inst] == true
            if shouldBe and not data.isActive then
                local hl = inst:FindFirstChild("OT_HL")
                if not hl then hl = createHighlightFor(inst, false) end
                hl.Enabled = true
                data.highlight = hl
                data.isActive = true
            elseif not shouldBe and data.isActive then
                local hl = inst:FindFirstChild("OT_HL")
                if hl then hl.Enabled = false end
                data.highlight = nil
                data.isActive = false
            end
        end
    end
end

-- ============================================================
-- LIST LOOP
-- ============================================================
task.spawn(function()
    while true do
        task.wait(CONFIG.listUpdateEvery)
        updateOwnHighlight()

        local pp = getPlayerPos()
        if not pp then
            pcall(function()
                ListParagraph:Set({Title="Tracked (0)", Content="No position."})
            end)
        else
            local rows, count = {}, 0
            for inst, data in pairs(tracked) do
                count = count + 1
                local pos = getPosSize(inst)
                if pos then
                    table.insert(rows, {
                        name = tostring(data.displayName or inst.Name),
                        dist = (pos - pp).Magnitude,
                        active = data.isActive,
                    })
                end
            end
            table.sort(rows, function(a,b) return a.dist < b.dist end)
            local lines = {}
            local limit = math.min(#rows, 20)
            for i = 1, limit do
                local r = rows[i]
                local mark = r.active and "[ON] " or "[  ] "
                table.insert(lines, string.format("%s%s - %.0fm", mark, r.name, r.dist))
            end
            if #rows > limit then
                table.insert(lines, "... +"..(#rows-limit).." more")
            end
            if #lines == 0 then lines = {"Nothing tracked."} end
            local mode = getOwnVehicle() and " [FPV]" or ""
            pcall(function()
                ListParagraph:Set({
                    Title = "Tracked ("..count..")"..mode.." active<="..CONFIG.maxActiveHighlights,
                    Content = table.concat(lines,"\n"),
                })
            end)
        end
    end
end)

local newObjectsQueue = {}

workspace.DescendantAdded:Connect(function(obj)
    if obj:IsA("BasePart") or obj:IsA("Model") then
        if hasBlacklistedName(obj.Name) then return end
        table.insert(newObjectsQueue, obj)
    end
end)

task.spawn(function()
    while true do
        task.wait(CONFIG.descendantBatch)
        if #newObjectsQueue > 0 then
            local queue = newObjectsQueue
            newObjectsQueue = {}
            for _, obj in ipairs(queue) do
                if obj.Parent and not tracked[obj] and isLikelyDrone(obj) then
                    if trackAllFlying then
                        addTracked(obj, obj.Name)
                    else
                        for _, p in ipairs(patterns) do
                            if nameMatchesPattern(obj.Name, p) then
                                addTracked(obj, p.displayName or obj.Name)
                                break
                            end
                        end
                    end
                end
            end
        end
    end
end)

-- HEARTBEAT
-- ============================================================
local updateInterval = 1 / CONFIG.updateHz
local accumulator = 0

RunService.Heartbeat:Connect(function(dt)
    accumulator = accumulator + dt
    if accumulator < updateInterval then return end
    accumulator = 0
    updateInterval = 1 / CONFIG.updateHz
    local ok, err = pcall(refreshHighlights)
    if not ok then warn("[OT] refresh error: "..tostring(err)) end
end)
