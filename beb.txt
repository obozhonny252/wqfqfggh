-- ============================================================
-- CONFIG
-- ============================================================
local CONFIG = {
    minDistance   = 100,
    maxDistance   = 5000,
    startDistance = 1000,

    -- Ключевые слова для авто-скана. Ищем по ПОДСТРОКЕ в имени.
    -- Регистр не важен.
    droneKeywords = {
        -- из меню RECON игры
        "vespa", "vector", "varan", "fpv", "kp-1", "kp1", "predator",

        -- пропеллеры (главная фишка — есть на всех дронах)
        "propeller", "prop", "rotor", "blade", "engine",

        -- общие термины дронов
        "drone", "uav", "bpla", "recon",
        "shahed", "shahid", "geran", "lancet", "orlan",
        "kamikaze", "missile", "rocket", "aircraft", "plane",
    },

    flyMinY          = 10,    -- min высота (studs), чтобы считать объект летающим
    maxSizeMagnitude = 60,    -- max размер (studs), чтобы не рисовать ландшафт

    outlineColor        = Color3.fromRGB(255, 80, 80),
    outlineTransparency = 0,
    fillTransparency    = 1,
    showOutline         = true,

    updateHz = 20,
}

local currentDistance = CONFIG.startDistance
local patterns        = {}
local tracked         = {}
local scanRunning     = false

-- ============================================================
-- RAYFIELD UI
-- ============================================================
local Window = Rayfield:CreateWindow({
    Name = "Object Tracker",
    LoadingTitle = "Object Tracker",
    LoadingSubtitle = "SKYFIRE edition",
    ConfigurationSaving = { Enabled = false },
    KeySystem = false,
})

local MainTab = Window:CreateTab("Main", nil)

MainTab:CreateSection("Distance")

MainTab:CreateSlider({
    Name = "Max distance",
    Range = {CONFIG.minDistance, CONFIG.maxDistance},
    Increment = 50,
    Suffix = "m",
    CurrentValue = CONFIG.startDistance,
    Flag = "Distance",
    Callback = function(v) currentDistance = v end,
})

MainTab:CreateSection("Auto scan")

local ScanStatus = MainTab:CreateParagraph({
    Title = "Scan",
    Content = "Not scanned yet.",
})

local ExtraKeywords = MainTab:CreateInput({
    Name = "Additional keywords (comma separated)",
    PlaceholderText = "e.g. krot, ugv, tank",
    RemoveTextAfterFocusLost = false,
    Flag = "ExtraKeywords",
    Callback = function() end,
})

MainTab:CreateButton({
    Name = "Scan now",
    Callback = function()
        if scanRunning then return end
        scanRunning = true
        ScanStatus:Set({ Title = "Scan", Content = "Scanning..." })
        task.wait(0.05)

        -- собираем все ключевые слова
        local allKeywords = {}
        for _, k in ipairs(CONFIG.droneKeywords) do table.insert(allKeywords, k) end
        for w in string.gmatch(ExtraKeywords.Value or "", "[^,]+") do
            w = w:gsub("^%s+", ""):gsub("%s+$", ""):lower()
            if w ~= "" then table.insert(allKeywords, w) end
        end

        local function nameMatches(name)
            local l = name:lower()
            for _, kw in ipairs(allKeywords) do
                if string.find(l, kw, 1, true) then return true end
            end
            return false
        end

        local function getInfo(obj)
            if obj:IsA("BasePart") then
                return obj.Position, obj.Size.Magnitude
            elseif obj:IsA("Model") then
                local ok, cf, sz = pcall(function()
                    local c, s = obj:GetBoundingBox()
                    return c, s
                end)
                if ok and cf and sz then return cf.Position, sz.Magnitude end
                return obj:GetPivot().Position, 5
            end
            return nil, 0
        end

        local groups = {}
        local total  = 0

        for _, obj in ipairs(workspace:GetDescendants()) do
            if obj:IsA("BasePart") or obj:IsA("Model") then
                local pos, size = getInfo(obj)
                if pos and size > 0 and size <= CONFIG.maxSizeMagnitude
                   and pos.Y >= CONFIG.flyMinY
                   and nameMatches(obj.Name) then
                    local base = obj.Name:gsub("%d+$", ""):gsub("[_%-%.]%d+$", "")
                    if base ~= "" then
                        groups[base] = (groups[base] or 0) + 1
                        total = total + 1
                    end
                end
            end
        end

        -- добавляем найденные базы в patterns
        local lines = {}
        for base, count in pairs(groups) do
            local exists = false
            for _, p in ipairs(patterns) do
                if p.id == base then exists = true break end
            end
            if not exists then
                table.insert(patterns, { id = base, match = "prefix", displayName = base })
            end
            table.insert(lines, string.format("%s (%d)", base, count))
        end

        -- сразу трекаем всё по всем patterns
        for _, p in ipairs(patterns) do
            for _, obj in ipairs(workspace:GetDescendants()) do
                if (obj:IsA("BasePart") or obj:IsA("Model")) and not tracked[obj] then
                    local ok, match = pcall(function()
                        if p.match == "exact" then
                            return obj.Name == p.id
                        elseif p.match == "contains" then
                            return string.find(obj.Name:lower(), p.id:lower(), 1, true) ~= nil
                        else
                            return string.sub(obj.Name, 1, #p.id):lower() == p.id:lower()
                        end
                    end)
                    if ok and match then addTracked(obj, p.displayName or obj.Name) end
                end
            end
        end

        if #lines == 0 then
            ScanStatus:Set({
                Title = "Scan - nothing found",
                Content = "Try adding keywords above (e.g. propeller, rotor).",
            })
        else
            table.sort(lines)
            table.insert(lines, 1, "Total objects: " .. total)
            ScanStatus:Set({
                Title = "Scan - " .. #groups .. " groups found",
                Content = table.concat(lines, "\n"),
            })
        end

        scanRunning = false
    end,
})

MainTab:CreateSection("Manual add")

local ManualInput = MainTab:CreateInput({
    Name = "Add by name/prefix",
    PlaceholderText = "Geran, Varan, ...",
    RemoveTextAfterFocusLost = false,
    Flag = "ManualInput",
    Callback = function() end,
})

MainTab:CreateButton({
    Name = "Add pattern",
    Callback = function()
        local raw = (ManualInput.Value or ""):gsub("^%s+", ""):gsub("%s+$", "")
        if raw == "" then return end
        table.insert(patterns, { id = raw, match = "prefix", displayName = raw })
        for _, obj in ipairs(workspace:GetDescendants()) do
            if (obj:IsA("BasePart") or obj:IsA("Model")) and not tracked[obj] then
                if string.sub(obj.Name, 1, #raw):lower() == raw:lower() then
                    addTracked(obj, obj.Name)
                end
            end
        end
    end,
})

-- ============================================================
-- HIGHLIGHT + TRACKER
-- ============================================================
local drawGui = Instance.new("ScreenGui")
drawGui.Name = "ObjectTracker_HL"
drawGui.ResetOnSpawn = false
drawGui.IgnoreGuiInset = true
drawGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
pcall(function()
    drawGui.Parent = game:GetService("CoreGui")
end)
if not drawGui.Parent then
    drawGui.Parent = LocalPlayer:WaitForChild("PlayerGui")
end

local function createHighlight(instance)
    if not CONFIG.showOutline then return nil end
    local hl = Instance.new("Highlight")
    hl.Name = "OT_HL"
    hl.Adornee = instance
    hl.FillTransparency = CONFIG.fillTransparency
    hl.OutlineColor = CONFIG.outlineColor
    hl.OutlineTransparency = CONFIG.outlineTransparency
    hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    hl.Parent = drawGui
    return hl
end

function addTracked(instance, displayName)
    if tracked[instance] then return false end
    local count = 0
    for _ in pairs(tracked) do count = count + 1 end
    if count >= 300 then return false end
    tracked[instance] = {
        displayName = displayName or instance.Name,
        highlight   = createHighlight(instance),
    }
    return true
end

-- ============================================================
-- LIST UPDATE
-- ============================================================
task.spawn(function()
    while true do
        task.wait(0.6)
        local char = LocalPlayer.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        if not root then
            ListParagraph:Set({ Title = "Tracked (0)", Content = "No character." })
        else
            local playerPos = root.Position
            local rows, count = {}, 0

            for inst, data in pairs(tracked) do
                if not inst.Parent then
                    if data.highlight then data.highlight:Destroy() end
                    tracked[inst] = nil
                else
                    count = count + 1
                    local pos
                    if inst:IsA("BasePart") then
                        pos = inst.Position
                    elseif inst:IsA("Model") then
                        local ok, cf = pcall(function() return inst:GetBoundingBox() end)
                        pos = ok and cf.Position or inst:GetPivot().Position
                    else
                        pos = Vector3.new()
                    end
                    table.insert(rows, { name = data.displayName,
                                         dist = (pos - playerPos).Magnitude })
                end
            end

            table.sort(rows, function(a, b) return a.dist < b.dist end)

            local lines = {}
            local limit = math.min(#rows, 20)
            for i = 1, limit do
                local r = rows[i]
                local mark = r.dist <= currentDistance and "[ON] " or "[  ] "
                table.insert(lines, string.format("%s%s - %.0fm", mark, r.name, r.dist))
            end
            if #rows > limit then
                table.insert(lines, "... +" .. (#rows - limit) .. " more")
            end
            if #lines == 0 then lines = { "Nothing tracked." } end

            ListParagraph:Set({
                Title = "Tracked (" .. count .. ")",
                Content = table.concat(lines, "\n"),
            })
        end
    end
end)

-- ============================================================
-- AUTO-PICKUP (новые объекты)
-- ============================================================
local function nameMatchesAnyPattern(name)
    for _, p in ipairs(patterns) do
        if p.match == "exact" then
            if name == p.id then return p end
        elseif p.match == "contains" then
            if string.find(name:lower(), p.id:lower(), 1, true) then return p end
        else
            if string.sub(name, 1, #p.id):lower() == p.id:lower() then return p end
        end
    end
    return nil
end

local function checkAndAdd(obj)
    if not (obj:IsA("BasePart") or obj:IsA("Model")) then return end
    if tracked[obj] then return end

    -- фильтр размера
    local size = 0
    if obj:IsA("BasePart") then
        size = obj.Size.Magnitude
    elseif obj:IsA("Model") then
        local ok, _, sz = pcall(function()
            local _, s = obj:GetBoundingBox()
            return _, s
        end)
        if ok and sz then size = sz.Magnitude end
    end
    if size > CONFIG.maxSizeMagnitude then return end

    local p = nameMatchesAnyPattern(obj.Name)
    if p then addTracked(obj, p.displayName or obj.Name) end
end

workspace.DescendantAdded:Connect(checkAndAdd)

-- ============================================================
-- UPDATE LOOP
-- ============================================================
local updateInterval = 1 / CONFIG.updateHz
local accumulator = 0

RunService.Heartbeat:Connect(function(dt)
    accumulator = accumulator + dt
    if accumulator < updateInterval then return end
    accumulator = 0
    for inst, data in pairs(tracked) do
        if not inst.Parent or not inst:IsDescendantOf(workspace) then
            if data.highlight then pcall(function() data.highlight:Destroy() end) end
            tracked[inst] = nil
        end
    end
end)

-- ============================================================
-- ИНИЦИАЛИЗАЦИЯ: pattern по всем ключевым словам
-- ============================================================
for _, kw in ipairs(CONFIG.droneKeywords) do
    table.insert(patterns, { id = kw, match = "contains", displayName = kw })
end

-- Автос‑скан при старте
task.spawn(function()
    task.wait(1)
    local found = {}
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") or obj:IsA("Model") then
            local size = obj:IsA("BasePart") and obj.Size.Magnitude or 5
            if size <= CONFIG.maxSizeMagnitude then
                local pos
                if obj:IsA("BasePart") then pos = obj.Position
                else
                    local ok, cf = pcall(function() return obj:GetBoundingBox() end)
                    pos = ok and cf.Position or obj:GetPivot().Position
                end
                if pos and pos.Y >= CONFIG.flyMinY then
                    local p = nameMatchesAnyPattern(obj.Name)
                    if p then
                        addTracked(obj, p.displayName or obj.Name)
                        found[p.id] = true
                    end
                end
            end
        end
    end
    local c = 0
    for _ in pairs(found) do c = c + 1 end
    ScanStatus:Set({
        Title = "Auto-scan on start",
        Content = "Found " .. c .. " keyword groups.",
    })
end)
