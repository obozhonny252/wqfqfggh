-- Загрузка Rayfield
local Rayfield = loadstring(game:HttpGet('https://sirius.menu/rayfield'))()

local Window = Rayfield:CreateWindow({
    Name = "Skyfire Line Cleaner",
    LoadingTitle = "Загрузка...",
    LoadingSubtitle = "Убираем линии сигнала",
    ConfigurationSaving = { Enabled = false },
    KeySystem = false
})

local MainTab = Window:CreateTab("Линии", 4483362458)

local isActive = false
local targets = {}

-- Функция поиска конкретных объектов
local function findLines()
    table.clear(targets)
    local playerGui = game.Players.LocalPlayer:FindFirstChild("PlayerGui")
    if not playerGui then return end
    
    local trv = playerGui:FindFirstChild("TRV")
    if trv then
        local droneHUD = trv:FindFirstChild("DroneHUD")
        if droneHUD then
            for _, name in ipairs({"Line1", "Line2", "Structure"}) do
                local obj = droneHUD:FindFirstChild(name)
                if obj then
                    table.insert(targets, obj)
                    if name == "Structure" then
                        for _, child in ipairs(obj:GetDescendants()) do
                            if child:IsA("ImageLabel") or child:IsA("Frame") then
                                table.insert(targets, child)
                            end
                        end
                    end
                end
            end
        end
    end
end

-- Принудительное скрытие (умное — трогает только то, что нужно)
local function forceHide()
    for _, obj in ipairs(targets) do
        if obj and obj.Parent then
            if obj.Visible then 
                pcall(function() obj.Visible = false end) 
            end
            if obj:IsA("ImageLabel") and obj.ImageTransparency < 1 then 
                pcall(function() obj.ImageTransparency = 1 end) 
            end
            if obj:IsA("Frame") and obj.BackgroundTransparency < 1 then 
                pcall(function() obj.BackgroundTransparency = 1 end) 
            end
        end
    end
end

-- Возврат в исходное состояние
local function restoreAll()
    for _, obj in ipairs(targets) do
        if obj and obj.Parent then
            pcall(function()
                obj.Visible = true
                if obj:IsA("ImageLabel") then obj.ImageTransparency = 0 end
                if obj:IsA("Frame") then obj.BackgroundTransparency = 0 end
            end)
        end
    end
end

-- ================= ИНТЕРФЕЙС =================

MainTab:CreateButton({
    Name = "Найти линии (Line1, Line2)",
    Callback = function()
        findLines()
        if isActive then forceHide() end
    end,
})

MainTab:CreateToggle({
    Name = "Убрать линии сигнала",
    CurrentValue = false,
    Flag = "DisableLines",
    Callback = function(Value)
        isActive = Value
        if isActive then
            if #targets == 0 then findLines() end
            forceHide()
        else
            restoreAll()
        end
    end,
})

-- ================= ОПТИМИЗИРОВАННЫЕ ЦИКЛЫ =================

-- Цикл 1: Обновление кэша (раз в 3 сек — только для поиска новых объектов)
task.spawn(function()
    while task.wait(3) do
        if isActive then findLines() end
    end
end)

-- Цикл 2: Принудительное скрытие (раз в 0.25 сек, но с проверками)
task.spawn(function()
    while task.wait(0.25) do
        if isActive and #targets > 0 then
            forceHide()
        end
    end
end)

-- ================= ВЫГРУЗКА =================
MainTab:CreateButton({
    Name = "Unload Script",
    Callback = function()
        isActive = false
        restoreAll()
        Rayfield:Destroy()
    end,
})
