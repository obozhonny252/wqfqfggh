-- ---------------- HELPERS (declared early) ----------------
function getPosSize(obj)
    if obj:IsA("BasePart") then
        return obj.Position, obj.Size.Magnitude
    elseif obj:IsA("Model") then
        local ok, cf, sz = pcall(function()
            local c, s = obj:GetBoundingBox(); return c, s
        end)
        if ok and cf and sz then return cf.Position, sz.Magnitude end
        return obj:GetPivot().Position, 5
    end
    return nil, 0
end

local function hasBlacklistedName(name)
    local l = name:lower()
    for _, w in ipairs(CONFIG.blacklistKeywords) do
        if string.find(l, w, 1, true) then return true end
    end
    return false
end

local function classAllowed(obj)
    return CONFIG.allowedClasses[obj.ClassName] == true
end

local function isLikelyDrone(obj)
    if not classAllowed(obj) then return false, nil end
    if hasBlacklistedName(obj.Name) then return false, nil end
    local pos, size = getPosSize(obj)
    if not pos then return false, nil end
    if size < CONFIG.minSizeMagnitude then return false, nil end
    if size > CONFIG.maxSizeMagnitude then return false, nil end
    if pos.Y < CONFIG.flyMinY then return false, nil end
    return true, pos, size
end

function nameMatchesPattern(name, p)
    if p.match == "exact" then
        return name == p.id
    elseif p.match == "contains" then
        return string.find(name:lower(), p.id:lower(), 1, true) ~= nil
    else
        return string.sub(name, 1, #p.id):lower() == p.id:lower()
    end
end

local function getAllKeywords()
    local list = {}
    for _, k in ipairs(CONFIG.droneKeywords) do table.insert(list, k) end
    for w in string.gmatch(ExtraKeywords.Value or "", "[^,]+") do
        w = w:gsub("^%s+", ""):gsub("%s+$", ""):lower()
        if w ~= "" then table.insert(list, w) end
    end
    return list
end

MainTab:CreateButton({
    Name = "Scan now",
    Callback = function()
        if scanRunning then return end
        scanRunning = true
        ScanStatus:Set({Title = "Scan", Content = "Scanning..."})
        task.wait(0.05)

        local kwList = getAllKeywords()
        local groups, total = {}, 0

        for _, obj in ipairs(workspace:GetDescendants()) do
            local ok, pos, size = isLikelyDrone(obj)
            if ok then
                local name = obj.Name:lower()
                local matched = false
                for _, kw in ipairs(kwList) do
                    if string.find(name, kw, 1, true) then matched = true break end
                end
                if matched then
                    local base = obj.Name:gsub("%d+$",""):gsub("[_%-%.]%d+$","")
                    if base ~= "" then
                        groups[base] = (groups[base] or 0) + 1
                        total = total + 1
                    end
                end
            end
        end

        local lines = {}
        for base, count in pairs(groups) do
            local exists = false
            for _, p in ipairs(patterns) do
                if p.id == base then exists = true break end
            end
            if not exists then
                table.insert(patterns, {id=base, match="prefix", displayName=base})
            end
            table.insert(lines, string.format("%s (%d)", base, count))
        end

        for _, p in ipairs(patterns) do
            for _, obj in ipairs(workspace:GetDescendants()) do
                if not tracked[obj] and isLikelyDrone(obj) and nameMatchesPattern(obj.Name, p) then
                    addTracked(obj, p.displayName or obj.Name)
                end
            end
        end

        if #lines == 0 then
            ScanStatus:Set({Title="Scan - nothing found",
                Content="Включи Track ALL или добавь keywords."})
        else
            table.sort(lines)
            table.insert(lines, 1, "Total: " .. total)
            ScanStatus:Set({Title="Scan - "..#groups.." groups",
                Content=table.concat(lines,"\n")})
        end
        scanRunning = false
    end,
})

local drawGui = Instance.new("ScreenGui")

local function createHighlight(inst)
    if not CONFIG.showOutline then return nil end
    local hl = Instance.new("Highlight")
    hl.Name = "OT_HL"
    hl.Adornee = inst
    hl.FillTransparency = CONFIG.fillTransparency
    hl.OutlineColor = CONFIG.outlineColor
    hl.OutlineTransparency = CONFIG.outlineTransparency
    hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    hl.Parent = drawGui
    return hl
end

function addTracked(inst, name)
    if tracked[inst] then return false end
    local count = 0
    for _ in pairs(tracked) do count = count + 1 end
    if count >= 300 then return false end
    tracked[inst] = {
        displayName = name or inst.Name,
        highlight   = createHighlight(inst),
    }
    return true
end

local function removeTracked(inst)
    local data = tracked[inst]
    if not data then return end
    if data.highlight then pcall(function() data.highlight:Destroy() end) end
    tracked[inst] = nil
end

-- ============================================================
-- IS DESTROYED (упавшие/мёртвые)
-- ============================================================
local function isDestroyed(obj)
    local pos = getPosSize(obj)
    if pos and pos.Y < CONFIG.groundY then return true end
    if obj:IsA("Model") then
        local hum = obj:FindFirstChildOfClass("Humanoid")
        if hum and hum.Health <= 0 then return true end
        local ok, v = pcall(function() return obj:GetAttribute("Destroyed") end)
        if ok and v == true then return true end
        local ok2, v2 = pcall(function() return obj:GetAttribute("dead") end)
        if ok2 and v2 == true then return true end
        if obj:HasTag("Destroyed") or obj:HasTag("Dead") then return true end
    end
    return false
end

-- ============================================================
-- LIST + CLEANUP
-- ============================================================
task.spawn(function()
    while true do
        task.wait(0.4)
        for inst in pairs(tracked) do
            if not inst.Parent or not inst:IsDescendantOf(workspace) then
                removeTracked(inst)
            elseif isDestroyed(inst) then
                removeTracked(inst)
            end
        end

        local char = LocalPlayer.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        if not root then
            ListParagraph:Set({Title="Tracked (0)", Content="No character."})
        else
            local pp = root.Position
            local rows, count = {}, 0
            for inst, data in pairs(tracked) do
                count = count + 1
                local pos = getPosSize(inst)
                if pos then
                    table.insert(rows, {
                        name = data.displayName,
                        dist = (pos - pp).Magnitude,
                    })
                end
            end
            table.sort(rows, function(a,b) return a.dist < b.dist end)
            local lines = {}
            local limit = math.min(#rows, 20)
            for i = 1, limit do
                local r = rows[i]
                local mark = r.dist <= currentDistance and "[ON] " or "[  ] "
                table.insert(lines, string.format("%s%s - %.0fm", mark, r.name, r.dist))
            end
            if #rows > limit then
                table.insert(lines, "... +"..(#rows-limit).." more")
            end
            if #lines == 0 then lines = {"Nothing tracked."} end
            ListParagraph:Set({Title="Tracked ("..count..")",
                Content=table.concat(lines,"\n")})
        end
    end
end)

-- ============================================================
-- AUTO-PICKUP
-- ============================================================
local function checkAndAdd(obj)
    if not (obj:IsA("BasePart") or obj:IsA("Model")) then return end
    if tracked[obj] then return end
    local ok = isLikelyDrone(obj)
    if not ok then return end

    if trackAllFlying then
        addTracked(obj, obj.Name)
        return
    end
    for _, p in ipairs(patterns) do
        if nameMatchesPattern(obj.Name, p) then
            addTracked(obj, p.displayName or obj.Name)
            return
        end
    end
end

workspace.DescendantAdded:Connect(checkAndAdd)

-- ============================================================
-- HEARTBEAT CLEANUP
-- ============================================================
local updateInterval = 1 / CONFIG.updateHz
local accumulator = 0

RunService.Heartbeat:Connect(function(dt)
    accumulator = accumulator + dt
    if accumulator < updateInterval then return end
    accumulator = 0

    for inst in pairs(tracked) do
        if not inst.Parent or not inst:IsDescendantOf(workspace) then
            removeTracked(inst)
        elseif isDestroyed(inst) then
            removeTracked(inst)
        end
    end

    if trackAllFlying then
        for _, obj in ipairs(workspace:GetDescendants()) do
            if not tracked[obj] and isLikelyDrone(obj) then
                addTracked(obj, obj.Name)
            end
        end
    end
end)

-- ============================================================
-- INIT
-- ============================================================
for _, kw in ipairs(CONFIG.droneKeywords) do
    table.insert(patterns, {id=kw, match="contains", displayName=kw})
end

task.spawn(function()
    task.wait(1)
    local found = {}
    for _, obj in ipairs(workspace:GetDescendants()) do
        if not tracked[obj] and isLikelyDrone(obj) then
            for _, p in ipairs(patterns) do
                if nameMatchesPattern(obj.Name, p) then
                    addTracked(obj, p.displayName or obj.Name)
                    found[p.id] = true
                    break
                end
            end
        end
    end
    local c = 0
    for _ in pairs(found) do c = c + 1 end
    ScanStatus:Set({Title="Auto-scan on start",
        Content="Found "..c.." groups."})
end)
