-- Загрузка Rayfield
local Rayfield = loadstring(game:HttpGet('https://sirius.menu/rayfield'))()

local Window = Rayfield:CreateWindow({
    Name = "Skyfire FPV Cleaner",
    LoadingTitle = "Загрузка...",
    LoadingSubtitle = "Убираем помехи дрона",
    ConfigurationSaving = { Enabled = true, FolderName = "SkyfireFPV", FileName = "FPVConfig" },
    KeySystem = false
})

local MainTab = Window:CreateTab("Помехи", 4483362458)

-- ================= НАСТРОЙКИ =================
local isActive = false
local alphaValue = 0
local targets = {}

-- Расширенные ключевые слова для поиска ВСЕХ видов помех, включая потерю сигнала
local keywords = {
    "noise", "jamming", "static", "glitch", "snow", "overlay", 
    "interference", "line", "distortion", "scanline", "strip", 
    "band", "video", "feed", "stripe", "artifact", "signal", 
    "loss", "dist", "far", "break", "corrupt"
}

-- ================= ФУНКЦИИ =================

local function findInterference()
    table.clear(targets)
    local playerGui = game.Players.LocalPlayer:FindFirstChild("PlayerGui")
    local camera = workspace.CurrentCamera
    local lighting = game.Lighting
    
    -- 1. Поиск в PlayerGui
    if playerGui then
        for _, obj in ipairs(playerGui:GetDescendants()) do
            if obj:IsA("ImageLabel") or obj:IsA("Frame") then
                local objName = obj.Name:lower()
                for _, kw in ipairs(keywords) do
                    if string.find(objName, kw) then
                        table.insert(targets, obj)
                        break
                    end
                end
            end
        end
    end
    
    -- 2. Поиск в CurrentCamera (иногда эффекты крепятся прямо к камере)
    if camera then
        for _, obj in ipairs(camera:GetChildren()) do
            if obj:IsA("ImageLabel") or obj:IsA("Frame") then
                local objName = obj.Name:lower()
                for _, kw in ipairs(keywords) do
                    if string.find(objName, kw) then
                        table.insert(targets, obj)
                        break
                    end
                end
            end
        end
    end
end

local function applySettings()
    -- Отключаем эффекты постобработки в Lighting, если они есть
    if game.Lighting:FindFirstChildOfClass("BlurEffect") then
        game.Lighting:FindFirstChildOfClass("BlurEffect").Enabled = not isActive
    end
    if game.Lighting:FindFirstChildOfClass("ColorCorrectionEffect") then
        game.Lighting:FindFirstChildOfClass("ColorCorrectionEffect").Enabled = not isActive
    end

    for _, obj in ipairs(targets) do
        if obj and obj.Parent then
            if isActive or alphaValue >= 1 then
                obj.Visible = false
                if obj:IsA("ImageLabel") then obj.ImageTransparency = 1 end
                if obj:IsA("Frame") then obj.BackgroundTransparency = 1 end
            else
                obj.Visible = true
                if obj:IsA("ImageLabel") then obj.ImageTransparency = alphaValue end
                if obj:IsA("Frame") then obj.BackgroundTransparency = alphaValue end
            end
            
            -- Применяем к дочерним элементам (например, полоски внутри объекта)
            for _, child in ipairs(obj:GetDescendants()) do
                if child:IsA("ImageLabel") then
                    child.ImageTransparency = (isActive or alphaValue >= 1) and 1 or alphaValue
                elseif child:IsA("Frame") then
                    child.BackgroundTransparency = (isActive or alphaValue >= 1) and 1 or alphaValue
                end
            end
        end
    end
end

-- ================= ИНТЕРФЕЙС =================

MainTab:CreateButton({
    Name = "Обновить поиск помех",
    Callback = function()
        findInterference()
        applySettings()
    end,
})

MainTab:CreateToggle({
    Name = "Убрать помехи полностью",
    CurrentValue = false,
    Flag = "DisableNoise",
    Callback = function(Value)
        isActive = Value
        applySettings()
    end,
})

MainTab:CreateSlider({
    Name = "Прозрачность помех",
    Range = {0, 1},
    Increment = 0.1,
    Suffix = "alpha",
    CurrentValue = 0,
    Flag = "NoiseAlpha",
    Callback = function(Value)
        alphaValue = Value
        isActive = false 
        applySettings()
    end,
})

-- ================= ФОНОВЫЕ ЦИКЛЫ =================

-- Поиск новых объектов (раз в 3 секунды, чтобы не лагало)
task.spawn(function()
    while task.wait(3) do
        findInterference()
    end
end)

-- Принудительное скрытие (раз в 0.15 секунды, чтобы игра не могла вернуть помехи)
task.spawn(function()
    while task.wait(0.15) do
        if isActive or alphaValue > 0 then
            applySettings()
        end
    end
end)

-- ================= ВЫГРУЗКА =================
MainTab:CreateButton({
    Name = "Unload Script",
    Callback = function()
        for _, obj in ipairs(targets) do
            if obj and obj.Parent then
                obj.Visible = true
                if obj:IsA("ImageLabel") then obj.ImageTransparency = 0 end
                if obj:IsA("Frame") then obj.BackgroundTransparency = 0 end
                for _, child in ipairs(obj:GetDescendants()) do
                    if child:IsA("ImageLabel") then child.ImageTransparency = 0 end
                    if child:IsA("Frame") then child.BackgroundTransparency = 0 end
                end
            end
        end
        Rayfield:Destroy()
        print("Skyfire FPV Cleaner выгружен.")
    end,
})
