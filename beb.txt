-- Загрузка Rayfield
local Rayfield = loadstring(game:HttpGet('https://sirius.menu/rayfield'))()

local Window = Rayfield:CreateWindow({
    Name = "Skyfire Line Cleaner",
    LoadingTitle = "Загрузка...",
    LoadingSubtitle = "Убираем линии сигнала",
    ConfigurationSaving = { Enabled = false },
    KeySystem = false
})

local MainTab = Window:CreateTab("Линии", 4483362458)

local isActive = false
local targets = {}

-- Расширенный список ключевых слов (тот, что у тебя работал)
local keywords = {
    "noise", "jamming", "static", "glitch", "snow", "overlay", 
    "interference", "line", "distortion", "scanline", "strip", 
    "band", "video", "feed", "stripe", "artifact", "signal", 
    "loss", "dist", "far", "break", "corrupt"
}

-- Функция поиска
local function findInterference()
    table.clear(targets)
    local playerGui = game.Players.LocalPlayer:FindFirstChild("PlayerGui")
    local camera = workspace.CurrentCamera
    
    local function scanContainer(container)
        if not container then return end
        for _, obj in ipairs(container:GetDescendants()) do
            if obj:IsA("ImageLabel") or obj:IsA("Frame") then
                local objName = obj.Name:lower()
                for _, kw in ipairs(keywords) do
                    if string.find(objName, kw) then
                        table.insert(targets, obj)
                        break
                    end
                end
            end
        end
    end

    scanContainer(playerGui)
    scanContainer(camera)
end

-- Принудительное скрытие (меняет только то, что реально нужно)
local function forceHide()
    for _, obj in ipairs(targets) do
        if obj and obj.Parent then
            pcall(function()
                if obj.Visible then obj.Visible = false end
                if obj:IsA("ImageLabel") and obj.ImageTransparency < 1 then 
                    obj.ImageTransparency = 1 
                end
                if obj:IsA("Frame") and obj.BackgroundTransparency < 1 then 
                    obj.BackgroundTransparency = 1 
                end
            end)
        end
    end
end

-- Возврат в исходное состояние
local function restoreAll()
    for _, obj in ipairs(targets) do
        if obj and obj.Parent then
            pcall(function()
                obj.Visible = true
                if obj:IsA("ImageLabel") then obj.ImageTransparency = 0 end
                if obj:IsA("Frame") then obj.BackgroundTransparency = 0 end
            end)
        end
    end
end

-- ================= ИНТЕРФЕЙС =================

MainTab:CreateButton({
    Name = "Обновить поиск помех",
    Callback = function()
        findInterference()
        if isActive then forceHide() end
    end,
})

MainTab:CreateToggle({
    Name = "Убрать помехи полностью",
    CurrentValue = false,
    Flag = "DisableLines",
    Callback = function(Value)
        isActive = Value
        if isActive then
            findInterference()
            forceHide()
        else
            restoreAll()
        end
    end,
})

-- ================= ОПТИМИЗИРОВАННЫЕ ЦИКЛЫ =================

-- Цикл 1: Поиск новых объектов (раз в 2 секунды — для подстраховки)
task.spawn(function()
    while task.wait(2) do
        if isActive then
            -- Проверяем, живы ли старые объекты
            local needsRefresh = false
            for _, obj in ipairs(targets) do
                if not obj or not obj.Parent then
                    needsRefresh = true
                    break
                end
            end
            if needsRefresh then
                findInterference()
            end
        end
    end
end)

-- Цикл 2: Принудительное скрытие (раз в 0.1 секунды, но с проверками — не жрёт ресурсы)
task.spawn(function()
    while task.wait(0.1) do
        if isActive and #targets > 0 then
            forceHide()
        end
    end
end)

-- ================= ВЫГРУЗКА =================
MainTab:CreateButton({
    Name = "Unload Script",
    Callback = function()
        isActive = false
        restoreAll()
        Rayfield:Destroy()
    end,
})
