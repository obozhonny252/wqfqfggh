-- Загрузка Rayfield
local Rayfield = loadstring(game:HttpGet('https://sirius.menu/rayfield'))()

-- Создание окна
local Window = Rayfield:CreateWindow({
    Name = "Skyfire FPV Cleaner",
    LoadingTitle = "Загрузка...",
    LoadingSubtitle = "Убираем помехи дрона",
    ConfigurationSaving = {
        Enabled = true,
        FolderName = "SkyfireFPV",
        FileName = "FPVConfig"
    },
    KeySystem = false
})

local MainTab = Window:CreateTab("Помехи", 4483362458)

-- ================= НАСТРОЙКИ =================
local isActive = false       -- Включен ли режим удаления помех
local alphaValue = 0         -- Прозрачность (0 - видно, 1 - невидимо)
local targets = {}           -- Кэш найденных объектов
local keywords = {"noise", "jamming", "static", "glitch", "snow", "overlay", "interference"} -- Ключевые слова для поиска

-- ================= ФУНКЦИИ =================

-- Функция поиска всех возможных помех во всем PlayerGui
local function findInterference()
    table.clear(targets)
    local playerGui = game.Players.LocalPlayer:FindFirstChild("PlayerGui")
    if not playerGui then return end
    
    for _, obj in ipairs(playerGui:GetDescendants()) do
        if obj:IsA("ImageLabel") or obj:IsA("Frame") then
            for _, kw in ipairs(keywords) do
                if string.find(obj.Name:lower(), kw) then
                    table.insert(targets, obj)
                    break
                end
            end
        end
    end
    print("Найдено объектов помех:", #targets)
end

-- Функция применения настроек к объектам
local function applySettings()
    for _, obj in ipairs(targets) do
        if obj and obj.Parent then
            -- Если ползунок на максимум или включен тумблер - скрываем
            if isActive or alphaValue >= 1 then
                obj.Visible = false
            else
                obj.Visible = true
                if obj:IsA("ImageLabel") then
                    obj.ImageTransparency = alphaValue
                elseif obj:IsA("Frame") then
                    obj.BackgroundTransparency = alphaValue
                end
            end
            
            -- Применяем к дочерним элементам (например, папка Images внутри Noise)
            for _, child in ipairs(obj:GetChildren()) do
                if child:IsA("ImageLabel") then
                    child.ImageTransparency = alphaValue
                elseif child:IsA("Frame") then
                    child.BackgroundTransparency = alphaValue
                end
            end
        end
    end
end

-- ================= ИНТЕРФЕЙС =================

-- Кнопка обновления поиска (если вы сели в новый дрон)
MainTab:CreateButton({
    Name = "Обновить поиск помех",
    Callback = function()
        findInterference()
        applySettings()
    end,
})

-- Переключатель для полного скрытия
MainTab:CreateToggle({
    Name = "Убрать помехи полностью",
    CurrentValue = false,
    Flag = "DisableNoise",
    Callback = function(Value)
        isActive = Value
        applySettings()
    end,
})

-- Ползунок прозрачности
MainTab:CreateSlider({
    Name = "Прозрачность помех",
    Range = {0, 1},
    Increment = 0.1,
    Suffix = "alpha",
    CurrentValue = 0,
    Flag = "NoiseAlpha",
    Callback = function(Value)
        alphaValue = Value
        isActive = false -- Если двигаем ползунок, выключаем тумблер
        applySettings()
    end,
})

-- ================= ФОНОВЫЙ ЦИКЛ (БЕЗ ЛАГОВ) =================
-- Этот цикл работает каждые 0.2 секунды (а не каждый кадр), чтобы игра не могла вернуть помехи обратно
task.spawn(function()
    while task.wait(0.2) do
        if isActive or alphaValue > 0 then
            -- Если игра пересоздала объекты, находим их заново (редко, но бывает)
            if #targets == 0 then
                findInterference()
            end
            
            -- Принудительно применяем настройки
            for _, obj in ipairs(targets) do
                if obj and obj.Parent then
                    if isActive then
                        obj.Visible = false
                    elseif alphaValue > 0 then
                        if obj:IsA("ImageLabel") then obj.ImageTransparency = alphaValue end
                        if obj:IsA("Frame") then obj.BackgroundTransparency = alphaValue end
                    end
                end
            end
        end
    end
end)

-- ================= ВЫГРУЗКА =================
MainTab:CreateButton({
    Name = "Unload Script",
    Callback = function()
        -- 1. Возвращаем все в исходное состояние
        for _, obj in ipairs(targets) do
            if obj and obj.Parent then
                obj.Visible = true
                if obj:IsA("ImageLabel") then obj.ImageTransparency = 0 end
                if obj:IsA("Frame") then obj.BackgroundTransparency = 0 end
                for _, child in ipairs(obj:GetChildren()) do
                    if child:IsA("ImageLabel") then child.ImageTransparency = 0 end
                    if child:IsA("Frame") then child.BackgroundTransparency = 0 end
                end
            end
        end
        
        -- 2. Уничтожаем интерфейс
        Rayfield:Destroy()
        print("Skyfire FPV Cleaner выгружен, помехи возвращены.")
    end,
})
