-- Загрузка Rayfield
local Rayfield = loadstring(game:HttpGet('https://sirius.menu/rayfield'))()

-- Создание окна
local Window = Rayfield:CreateWindow({
    Name = "Skyfire FPV Cleaner",
    LoadingTitle = "Загрузка...",
    LoadingSubtitle = "Убираем помехи дрона",
    ConfigurationSaving = {
        Enabled = true,
        FolderName = "SkyfireFPV",
        FileName = "FPVConfig"
    },
    KeySystem = false
})

-- Создание вкладки
local MainTab = Window:CreateTab("Помехи", 4483362458)

-- Функция для поиска объектов помех (Noise и Jamming)
local function getNoiseObjects()
    local objects = {}
    local playerGui = game.Players.LocalPlayer:FindFirstChild("PlayerGui")
    if not playerGui then return objects end
    
    local trv = playerGui:FindFirstChild("TRV")
    if trv then
        local droneHUD = trv:FindFirstChild("DroneHUD")
        if droneHUD then
            local noise = droneHUD:FindFirstChild("Noise")
            local jamming = droneHUD:FindFirstChild("Jamming")
            
            if noise then table.insert(objects, noise) end
            if jamming then table.insert(objects, jamming) end
        end
    end
    return objects
end

-- Переключатель для полного скрытия помех
MainTab:CreateToggle({
    Name = "Убрать помехи полностью",
    CurrentValue = false,
    Flag = "DisableNoise",
    Callback = function(Value)
        local objects = getNoiseObjects()
        for _, obj in ipairs(objects) do
            if obj:IsA("ImageLabel") or obj:IsA("Frame") or obj:IsA("GuiObject") then
                obj.Visible = not Value
            end
        end
    end,
})

-- Ползунок для регулировки прозрачности помех
MainTab:CreateSlider({
    Name = "Прозрачность помех",
    Range = {0, 1},
    Increment = 0.1,
    Suffix = "alpha",
    CurrentValue = 0,
    Flag = "NoiseAlpha",
    Callback = function(Value)
        local objects = getNoiseObjects()
        for _, obj in ipairs(objects) do
            if obj:IsA("ImageLabel") then
                obj.ImageTransparency = Value
            elseif obj:IsA("Frame") then
                obj.BackgroundTransparency = Value
            end
            -- Пробегаемся по дочерним элементам (папка Images)
            for _, child in ipairs(obj:GetDescendants()) do
                if child:IsA("ImageLabel") then
                    child.ImageTransparency = Value
                elseif child:IsA("Frame") then
                    child.BackgroundTransparency = Value
                end
            end
        end
    end,
})

-- Кнопка выгрузки скрипта (Unload Script)
MainTab:CreateButton({
    Name = "Unload Script",
    Callback = function()
        -- 1. Возвращаем игру в исходное состояние (включаем помехи обратно)
        local objects = getNoiseObjects()
        for _, obj in ipairs(objects) do
            if obj:IsA("ImageLabel") or obj:IsA("Frame") or obj:IsA("GuiObject") then
                obj.Visible = true
            end
            if obj:IsA("ImageLabel") then
                obj.ImageTransparency = 0
            elseif obj:IsA("Frame") then
                obj.BackgroundTransparency = 0
            end
            for _, child in ipairs(obj:GetDescendants()) do
                if child:IsA("ImageLabel") then
                    child.ImageTransparency = 0
                elseif child:IsA("Frame") then
                    child.BackgroundTransparency = 0
                end
            end
        end
        
        -- 2. Полностью удаляем интерфейс Rayfield
        Rayfield:Destroy()
        print("Skyfire FPV Cleaner успешно выгружен.")
    end,
})
